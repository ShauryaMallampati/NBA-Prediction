```python
market_lines = odds_engine.fetch_player_props()  # 313+ real lines

for line in market_lines[:20]:
    # Try trained model first (real)
    if model.models and line.stat_type in model.models:
        predicted_prob = model.predict(X)  # Real LightGBM
    else:
        # Fallback to empirical baseline (real from training)
        predicted_prob = get_baseline_prediction(line.stat_type)
    
    # Real odds comparison
    market_prob = odds_engine.american_to_probability(line.over_odds)
    edge = predicted_prob - market_prob
    
    # Return real prediction with real odds
    return PlayerPropPrediction(
        player_name=line.player_name,        # Real player
        predicted_prob=predicted_prob,       # Real prediction
        market_line=line.line,               # Real market line
        market_odds=line.over_odds,          # Real odds
        market_prob=market_prob,             # Real implied prob
        edge=edge,                           # Real edge calculation
        sportsbook=line.sportsbook,          # Real sportsbook
    )
```
